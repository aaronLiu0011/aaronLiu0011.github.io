package trading_system;

import java.io.IOException;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) throws IOException {
        System.out.println("株式取引管理システムを開始します。");
        Scanner sc = new Scanner(System.in);

        while(true){
            ShowMenu();
            String c = sc.next().toUpperCase();

            switch(c) {
                case "A":
                    System.out.println("銘柄マスタを表示します。");
                    showStock.run();
                    System.out.println("---");
                    continue;
                case "B":
                    System.out.println("新規株式銘柄マスタを登録します。");
                    registerStock.run();
                    System.out.println("---");
                    continue;
                case "Q":
                    System.out.println("アプリケーションを終了しました。");
                    return;
                default:
                    System.out.printf("\"%s\" に対応するメニューは存在しません。%n", c);
            }
            System.out.println("---");
        }

    }

    private static void ShowMenu() {
        System.out.println("操作するメニューを選んでください。");
        System.out.println("    A: 銘柄マスター一覧表示");
        System.out.println("    B: 銘柄マスター新規登録");
        System.out.println("    Q: アプリケーションを終了する");
        System.out.print("入力してください：");
    }
}

package trading_system;

import java.util.Set;

// for single stock
public class stock {
    private String ticker;
    private String name;
    private String market;
    private long shares;

    public stock() {
    }

    public stock(String ticker, String name, String market, long shares) {
        this.ticker = ticker;
        this.name = name;
        this.market = market;
        this.shares = shares;
    }

    public String getTicker() {
        return ticker;
    }

    public String getName() {
        return name;
    }

    public String getMarket() {
        return market;
    }

    public long getShares() {
        return shares;
    }

    public void setTicker(String ticker) {
        this.ticker = ticker;
    }

    public void setName(String name) {
        this.name = name;
    }

    public void setMarket(String market) {
        this.market = market;
    }

    public void setShares(long shares) {
        this.shares = shares;
    }

    public String[] toArray() {
        return new String[]{ticker, name, market, String.valueOf(shares)};
    }

    // Ticker rules:
    //   1. 4 chars
    //   2. 1st and 3rd are digital
    //   3. 2nd and 4th are digital or allowed char
    public static final Set<Character> allowed_char = Set.of('A','C','D','F','G','H','J','K','L','M','N','P','R','S','T','U','W','X','Y');
    public static boolean isValidTicker(String ticker) {
        if (ticker.length() != 4) return false;
        if (!Character.isDigit(ticker.charAt(0))) return false;
        if (!Character.isDigit(ticker.charAt(2))) return false;
        if (!allowed_char.contains(ticker.charAt(1)) && !Character.isDigit(ticker.charAt(1))) return false;
        if (!allowed_char.contains(ticker.charAt(3)) && !Character.isDigit(ticker.charAt(3))) return false;
        return true;
    }

    // Name rules:
    public static boolean isValidName(String name){
        String trimmed = name.trim().replaceAll("^[\u3000\t]+|[\u3000\t]+$", "");
        if (trimmed.isEmpty()) return false;
        return trimmed.matches("[a-zA-Z0-9 .()]+");
    }

    // Market rules:
    public static final Set<String> allowed_market = Set.of("プライム", "スタンダード", "グロース", "Prime", "Standard", "Growth", "P", "S", "G");
    public static boolean isValidMarket(String market){
        market = market.substring(0, 1).toUpperCase() + market.substring(1);
        return (allowed_market.contains(market));
    }

    // Shares rule:
    public static boolean isValidShares(long shares){
        return shares >= 0 && shares <= 999_999_999_999L;
    }
}

stockManager

package trading_system;

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.ArrayList;
import java.util.List;

// to manage stock list
public class stockManager {
    private final List<stock> stockList = new ArrayList<>();

    public void loadFromCSV(String csvPath) throws IOException {
        stockList.clear();

        try (BufferedReader reader = Files.newBufferedReader(Paths.get(csvPath))) {
            reader.readLine();
            String line;
            while ((line = reader.readLine()) != null) {
                String[] parts = line.split(",");
                if (parts.length < 4) continue;
                stockList.add(
                        new stock(parts[0], parts[1], parts[2], Long.parseLong(parts[3]))
                );
            }

        }
    }

    public void saveToCSV(String csvPath) throws IOException {
        try (BufferedWriter writer = Files.newBufferedWriter(Paths.get(csvPath))) {
            writer.write("ticker,product_name,market,shares_issued\n");
            for (stock s : stockList) {
                writer.write(String.join(",", s.toArray()));
                writer.write("\n");
            }
        }
    }

    public void addStock(stock stock) {
        for (stock s : stockList) {
            if (s.getTicker().equals(stock.getTicker())) {
                System.err.println(stock.getTicker() + " already exists");
                return;
            }
        }
        stockList.add(stock);
    }

    public List<stock> getStockList() {
        return stockList;
    }
}

showStock

package trading_system;

import java.io.*;
import java.util.*;

public class showStock {
    public static void run() throws IOException {
        stockManager manager = new stockManager();
        manager.loadFromCSV("data/company_list.csv");
        List<stock> stockList = manager.getStockList();
        String[] header = {"Ticker", "Product Name", "Market", "Shares Issued"};

        //
        int columnsCount = header.length;
        int[] columnWidths = new int[columnsCount];

        // formate stock list
        for (stock stock : stockList) {
            stock.setName(truncation(stock.getName(),28));
            stock.setMarket(marketType(stock.getMarket()));
        }

        // calculate the max length of each column
        for (int i = 0; i < columnsCount; i++) {
            columnWidths[i] = header[i].length();
        }

        for (stock stock : stockList) {
            columnWidths[0] = Math.max(stock.getTicker().length(), columnWidths[0]);
            columnWidths[1] = Math.max(stock.getName().length(), columnWidths[1]);
            columnWidths[2] = Math.max(stock.getMarket().length(), columnWidths[2]);
            columnWidths[3] = Math.max(String.format("%,d", stock.getShares()).length(), columnWidths[3]);
        }

        int colWidthSum = 0;
        for (int i = 0; i < columnsCount; i++) {
            colWidthSum += columnWidths[i];
        }

        // print
        System.out.println(divider(columnsCount, colWidthSum, "="));
        System.out.println(formattedHeader(header, columnWidths));
        System.out.println(headerDivider(columnWidths, "-"));
        for (stock stock : stockList) {
            System.out.println(formattedRow(stock, columnWidths));
        }
        System.out.println(divider(columnsCount, colWidthSum, "="));

    }

    public static void main(String[] args) throws IOException {
        run();
    }

    // ===============================================================
    // market type
    public static String marketType(String arg) {
        arg = arg.substring(0, 1).toUpperCase() + arg.substring(1);
        switch (arg) {
            case "P": return "Prime";
            case "S": return "Standard";
            case "G": return "Growth";
            default: return arg;
        }
    }

    // Truncation
    public static String truncation(String arg, int maxWidth) {
        if (arg.length() <= maxWidth) return arg;
        return arg.substring(0, maxWidth-1).trim() + "...";
    }

    // format each row
    public static String formattedHeader (String[] header, int[] columnWidths) {
        StringBuilder sb = new StringBuilder();
        sb.append("|");
        for (int i = 0; i < header.length; i++) {
            sb.append(" ").append(String.format("%-" + columnWidths[i] + "s", header[i])).append(" |");
        }
        return sb.toString();
    }

    public static String formattedRow (stock s, int[] columnWidths) {
        StringBuilder sb = new StringBuilder();
        // first col
        sb.append("|");
        sb.append("  ").append(s.getTicker()).append("  |");
        //middle col
        sb.append(" ").append(String.format("%-" + columnWidths[1] + "s", truncation(s.getName(),28))).append(" |");
        sb.append(" ").append(String.format("%-" + columnWidths[2] + "s", s.getMarket())).append(" |");
        //last col
        sb.append(" ").append(String.format("%" + columnWidths[3] + "s", String.format("%,d", s.getShares()))).append(" |");
        return sb.toString();
    }

    // seperator
    public static String divider (int columnsCount, int colWidthSum, String separator) {
        StringBuilder sb = new StringBuilder();
        sb.append("|");
        for (int i = 0; i < colWidthSum + (columnsCount*3-1); i++) {
            sb.append(separator);
        }
        sb.append("|");
        return sb.toString();
    }

    public static String headerDivider (int[] columnWidths, String separator) {
        StringBuilder sb = new StringBuilder();
        sb.append("|");
        sb.append(separator.repeat(columnWidths[0]+2));
        sb.append("+");
        for (int i = 1; i < columnWidths.length-1; i++) {
            sb.append(separator.repeat(columnWidths[i]+2));
            sb.append("+");
        }
        sb.append(separator.repeat(columnWidths[columnWidths.length-1]+2));
        sb.append("|");
        return sb.toString();
    }

}

registerStock

package trading_system;

import java.io.IOException;
import java.util.List;
import java.util.Scanner;

public class registerStock {
    public static void run() throws IOException {
        stockManager manager = new stockManager();
        manager.loadFromCSV("data/company_list.csv");

        stock newStock = promptNewStock(manager.getStockList());
        manager.addStock(newStock);

        manager.saveToCSV("data/company_list.csv");
    }

    public static void main(String[] args) throws IOException {
        run();
    }

    public static stock promptNewStock(List<stock> existingStocks) throws IOException {
        Scanner sc = new Scanner(System.in);

        String name = null;
        String ticker = null;
        String market = null;
        long shares = 0;

        // 銘柄コード
        while (true) {
            System.out.print("銘柄コード> ");
            ticker = sc.nextLine().toUpperCase();

            boolean isDuplicate = false;
            for (stock s : existingStocks) {
                if (s.getTicker().equals(ticker)) {
                    System.out.println("この銘柄は既に登録されています。");
                    isDuplicate = true;
                    break;
                }
            }
            if (isDuplicate) continue; // jump to the beginning of while
            if (stock.isValidTicker(ticker)) break; // stop looping
            else System.out.println("銘柄コードが規則に従っていません。");
        }

        // 銘柄名
        while (true) {
            System.out.print("銘柄名> ");
            name = sc.nextLine().trim();
            if (stock.isValidName(name)) break;
            else System.out.println("銘柄名が規則に従っていません。");
        }

        // 上場市場
        while (true) {
            System.out.print("上場市場> ");
            market = sc.next();
            if (stock.isValidMarket(market)) break;
            else System.out.println("上場市場が規則に従っていません。");
        }

        // 発行済み株式数
        while (true) {
            System.out.print("発行済み株式数> ");
            shares = sc.nextLong();
            if (stock.isValidShares(shares)) break;
            else System.out.println("株式数が規則に従っていません。");
        }

        System.out.println(name + " を新規銘柄として登録しました。");
        return new stock(ticker, name, market, shares);
    }
}
